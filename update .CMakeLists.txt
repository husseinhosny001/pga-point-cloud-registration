# ==============================================================================
# Unit Testing Setup (GoogleTest)
# ==============================================================================
option(BUILD_TESTS "Build GoogleTest unit test suite" ON)

if(BUILD_TESTS)
    enable_testing()

    include(FetchContent)
    FetchContent_Declare(
        googletest
        URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz
    )
    # For Windows: Prevent overriding the parent project's compiler/linker settings
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)

    # Define Unit Test Executable
    add_executable(pga_unit_tests tests/test_motor.cpp)
    target_link_libraries(pga_unit_tests PRIVATE pga_icp_core gtest_main)

    include(GoogleTest)
    gtest_discover_tests(pga_unit_tests)
endif()
